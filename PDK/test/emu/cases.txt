# Hand-computed expectations for the PDK module (flags: ZF=1 CF=2 AC=4 OV=8; CF/AC are carries for additions and
# borrows for subtractions).  Worked out from the Free PDK instruction tables and the Padauk datasheets,
# never copied from the module's output.  See emu-test.sh for the format.

############################ PDK14 (PMS132 / PMS152) ############################
case add_basic isa 14
asm MOV A,#0x12 ; ADD A,#0x34
code 2F12 2834
expect A 0x46
expect flag 0
end

case add_signed_overflow isa 14
# 0x7f + 1 = 0x80: AC (0xf+1 > 0xf) and OV, no carry, not zero
asm MOV A,#0x7f ; ADD A,#0x01
code 2F7F 2801
expect A 0x80
expect flag 0xC
end

case add_carry_out isa 14
# 0xff + 1 = 0x100: A=0, ZF CF AC
asm MOV A,#0xff ; ADD A,#0x01
code 2FFF 2801
expect A 0
expect flag 0x7
end

case sub_borrow isa 14
# 0 - 1: A=0xff, borrow (CF), aux borrow (AC), no overflow
asm MOV A,#0x00 ; SUB A,#0x01
code 2F00 2901
expect A 0xFF
expect flag 0x6
end

case sub_equal isa 14
asm MOV A,#0x05 ; SUB A,#0x05
code 2F05 2905
expect A 0
expect flag 0x1
end

case sub_signed_overflow isa 14
# 0x80 - 1 = 0x7f: AC borrow (0 < 1) and OV
asm MOV A,#0x80 ; SUB A,#0x01
code 2F80 2901
expect A 0x7F
expect flag 0xC
end

case subc_uses_borrow isa 14
# 0-1 sets CF; then 5 - CF = 4, no flags
asm MOV A,#0x00 ; SUB A,#0x01 ; MOV A,#0x05 ; SUBC A
code 2F00 2901 2F05 0061
expect A 4
expect flag 0
end

case addc_uses_carry isa 14
asm MOV A,#0xff ; ADD A,#0x01 ; MOV A,#0x10 ; ADDC A
code 2FFF 2801 2F10 0060
expect A 0x11
expect flag 0
end

case and_zero isa 14
asm MOV A,#0xf0 ; AND A,#0x0f
code 2FF0 2C0F
expect A 0
expect flag 1
end

case src_rotates_through_carry isa 14
# CF=1 from 0xff+1 (flags 7); SRC A: A = 0x81, CF <- old bit0 = 0, ZF/AC kept: flag 5
asm MOV A,#0xff ; ADD A,#0x01 ; MOV A,#0x02 ; SRC A
code 2FFF 2801 2F02 006C
expect A 0x81
expect flag 5
end

case sl_sets_carry isa 14

end

case swap_nibbles isa 14

end

case neg_then_not isa 14
asm MOV A,#0x01 ; NEG A ; NOT A

expect flag 1
end

case inc_memory isa 14
asm MOV A,#0x55 ; MOV 0x0010,A ; INC 0x0010

expect flag 0
end

case xch_memory isa 14
asm MOV A,#0x01 ; MOV 0x0008,A ; MOV A,#0x02 ; XCH 0x0008
code 2F01 0B88 2F02 1388
expect ram:8 2
expect A 1
end

case dzsn_skips_when_zero isa 14
# [4]=1; DZSN [4] -> 0 and skips "MOV A,#0x99"; flag ZF (1-1, no borrow)
asm MOV A,#0x01 ; MOV 0x0004,A ; DZSN 0x0004 ; MOV A,#0x99 ; MOV A,#0x77
code 2F01 0B84 1184 2F99 2F77
expect A 0x77
expect ram:4 0
expect flag 1
end

case ceqsn_not_equal_runs_next isa 14
# 3 - 4: borrow, aux borrow; not equal -> no skip, A=9

code 2F03 2A04 2F09
expect A 9
expect flag 6
end

case flag_is_io_register isa 14
# MOV flag,A writes the flag bits themselves (ZF CF AC OV); SET1 flag.0 sets ZF
asm MOV A,#0x0a ; MOV flag,A ; SET1 flag.0
code 2F0A 0180 1E00
expect flag 0x0B
expect A 0x0A
end

case mov_a_io_updates_zf isa 14
# MOV A,flag with flag=5: A=5, ZF recomputed from A (non-zero -> cleared): flag 4

expect A 5
expect flag 4
end

case call_and_ret isa 14
# CALL pushes the word address of the next instruction (1), RET pops it
asm CALL 0x0004 ; MOV A,#0x02 ; GOTO 0x0006 ; MOV A,#0x00 ; MOV A,#0x01 ; RET
code 3804 2F02 3006 2F00 2F01 007A
set sp 0x20
until 6
expect A 2
expect sp 0x20
expect ram:20 1
expect ram:21 0
expect PC 6
end

end

case pcadd_adds_to_its_own_address isa 14

end

case nadd_is_add_of_negated_a isa 14
# A <- M + NEG(A) = 10 - 3 = 7; addition 0x0a + 0xfd: CF, AC; no ZF/OV
asm MOV A,#0x0a ; MOV 0x0005,A ; MOV A,#0x03 ; NADD A,0x0005
code 2F0A 0B85 2F03 0705
expect A 7
expect flag 6
end

case swapc_io_bit isa 14
# CF=1 (0xff+1) swaps with pa.3 (=0): pa=0x08, CF=0 -> flags 7 -> 5
asm MOV A,#0xff ; ADD A,#0x01 ; SWAPC pa.3
code 2FFF 2801 04D0
expect pa 0x08
expect flag 5
end

end
